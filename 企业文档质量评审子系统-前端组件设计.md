# 企业文档质量评审子系统 - 前端组件设计

## 1. 组件架构总览

基于现有Vue3 + Element Plus架构，采用Composition API和TypeScript：

```
src/views/review-system/
├── dashboard/                 # 评审总览仪表盘
├── task-management/          # 评审任务管理
├── expert-management/        # 专家管理
├── workflow-config/          # 流程配置
├── review-workspace/         # 评审工作台
├── issue-management/         # 问题管理
├── analytics/               # 统计分析
└── components/              # 共享组件
```

```vue
  })
})
```

```vue
    await updateWorkflow(workflow)

    await createWorkflow(workflow)
  }
  designerVisible.value = false
  fetchWorkflows()
}

onMounted(() => {
  fetchWorkflows()
})
</script>
```

### 4.1 useReviewWorkspace

```typescript
// composables/useReviewWorkspace.ts
import { ref, computed } from 'vue'
import { reviewApi } from '@/api/review'

export function useReviewWorkspace(taskId: number) {
  const currentTask = ref(null)
  const issues = ref([])
  const annotations = ref([])
  const selectedText = ref('')
  const aiAssistantVisible = ref(false)
  const loading = ref(false)

  const fetchTaskDetail = async () => {
    loading.value = true
    try {
      const response = await reviewApi.getTaskDetail(taskId)
      currentTask.value = response.data
    } finally {
      loading.value = false
    }
  }

  const toggleAIAssistant = () => {
    aiAssistantVisible.value = !aiAssistantVisible.value
  }

  const handleAddAnnotation = (annotation) => {
    annotations.value.push(annotation)
  }

  const handleTextSelection = (text) => {
    selectedText.value = text
  }

  const saveProgress = async () => {
    // 保存当前进度
  }

  const submitReview = async () => {
    // 提交评审
  }

  return {
    currentTask,
    issues,
    annotations,
    selectedText,
    aiAssistantVisible,
    loading,
    fetchTaskDetail,
    toggleAIAssistant,
    handleAddAnnotation,
    handleTextSelection,
    saveProgress,
    submitReview
  }
}
```
